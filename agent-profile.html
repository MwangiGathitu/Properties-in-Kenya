async function load() { console.log("=== 🕵️ AGENT PROFILE DEBUG START ==="); const params = new URLSearchParams(location.search); const id = params.get('id'); const slug = params.get('slug'); console.log("1. URL ID parameter:", id); console.log("2. URL Slug parameter:", slug); // If no ID or slug, show all agents if (!id && !slug) { console.log("⚠️ No ID or Slug found in URL. Showing all agents."); await showAllAgents(); return; } let profileData = null; let queryError = null; if (id) { console.log("3. Querying database by ID:", id); const res = await sb.from('profiles').select('*').eq('id', id).maybeSingle(); profileData = res.data; queryError = res.error; } else if (slug) { console.log("3. Querying database by Slug:", slug); const res = await sb.from('profiles').select('*').eq('slug', slug).maybeSingle(); profileData = res.data; queryError = res.error; } console.log("4. Database Query Result Data:", profileData); console.log("5. Database Query Error:", queryError); if (!profileData) { console.log("❌ FAILED: Profile data is null. Falling back to all agents."); document.getElementById('agentName').textContent = 'Agent not found'; document.getElementById('bio').textContent = `System looked for slug: "${slug}" or id: "${id}" but found nothing in the database. Showing all agents below.`; await showAllAgents(); return; } console.log("✅ SUCCESS: Profile found! Rendering page..."); profile = profileData; // Load listings const { data: l } = await sb.from('properties').select('*') .eq('agent_id', profile.id).eq('approval_status', 'approved') .order('created_at', { ascending: false }); listings = l || []; // Load reviews const { data: r } = await sb.from('reviews').select('*') .eq('agent_id', profile.id).eq('is_approved', true) .order('created_at', { ascending: false }); reviews = r || []; // Increment views sb.from('profiles').update({ profile_views: (profile.profile_views || 0) + 1 }).eq('id', profile.id); render(); }
